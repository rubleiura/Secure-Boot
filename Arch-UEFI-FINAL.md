# 💡 Метод: Копируйте и вставляйте блоки команд по одному.

#-----------------------------------------------------------------
# 🔧 РЕДАКЦИЯ v3 — финальная, вычитанная по трём реальным установкам.
#
# ✅ Проверено на практике: установка в VirtualBox по SSH, Secure Boot
#    активируется сам после записи ключей, UKI грузится, снимки btrfs
#    появляются в меню мгновенно (path-юнит), plymouth (тема bgrt) работает.
#
# Ключевые отличия от версии с GRUB:
# • Загрузчик — systemd-boot + UKI (НЕ GRUB): при включённом Secure Boot GRUB
#   из Arch отказывается грузить модули без shim (shim_lock_verifier_init:
#   prohibited by secure boot policy) и падает в grub rescue.
# • Ядро — linux-lts; параметры ядра — в /etc/kernel/cmdline (не /etc/default/grub).
# • Secure Boot — sbctl: свои ключи + подпись systemd-boot, BOOTX64.EFI и UKI.
# • Снимки btrfs в меню загрузки — БЛОК 13A (замена grub-btrfs).
# • Откат через btrfs-assistant: генератор не стирает пункты-снимки (БЛОК 13A),
#   а если дерево .snapshots осталось в @_backup_* — БЛОК 13B вернёт его в корень.
# • Plymouth (тема bgrt — родная для Arch) настраивается сразу в chroot,
#   а тихая загрузка (quiet splash loglevel=3) — ТОЛЬКО в БЛОКЕ 20.
# • Исправлены: порядок mkinitcpio/sign, ParallelDownloads, sed по fstab,
#   пакеты (plocate/pciutils), NVIDIA-параметры, опечатки в блоках 14–18.
# • Корень монтируется ЯВНО по ИМЕНИ: subvol=/@ в fstab (БЛОК 9) и
#   rootflags=subvol=@ в параметрах ядра (БЛОК 12). Подтом по умолчанию
#   НЕ меняется: иначе «Восстановить копию» в btrfs-assistant (переименование
#   подтомов) оставит систему загруженной из backup-копии.
# • Права на ESP: fmask=0177,dmask=0077 (БЛОК 9) — иначе systemd-boot пишет
#   "random seed file is world accessible, which is a security hole".
#   Побочный эффект: читать /boot можно только через sudo.
# • modules-load.d — по одному модулю на строку (БЛОК 15).
# • Генератор меню отбирает ТОЛЬКО снимки корневого подтома (БЛОК 13A):
#   снимок @home в качестве корня не загружается.
# • quiet splash включается в БЛОКЕ 20, ПОСЛЕ первой успешной загрузки, —
#   иначе причину сбоя не увидеть.

def haversine(lat1, lon1, lat2, lon2):
    R = 6371.0
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    a = math.sin(dlat/2)**2 + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlon/2)**2
    return R * 2 * math.atan2(math.sqrt(a), math.sqrt(1-a))

def get_ip_location():
    apis = [
        ("https://ipapi.co/json/", "ipapi.co"),
        ("https://ipinfo.io/json", "ipinfo.io"),
    ]
    for api_url, api_name in apis:
        try:
            print(f"  🔄 Пробуем {api_name}...")
            with urllib.request.urlopen(api_url, timeout=5) as resp:
                data = json.loads(resp.read().decode())

                # ✅ ИСПРАВЛЕНО: Добавлена переменная data в условия
                if api_name == "ipapi.co":
                    if "error" in data:
                        continue
                    if "latitude" not in data or "longitude" not in data:
                        continue
                    lat = float(data.get("latitude", 0))
                    lon = float(data.get("longitude", 0))
                    country = data.get("country_name", "Unknown")

                elif api_name == "ipinfo.io":
                    if "bogon" in data:
                        continue
                    if "loc" not in data:
                        continue
                    loc = data.get("loc", "0,0").split(",")
                    lat = float(loc[0])
                    lon = float(loc[1])
                    country = data.get("country", "Unknown")
                else:
                    continue

                if lat != 0 and lon != 0:
                    print(f"  ✅ Использован API: {api_name}")
                    return lat, lon, country

        except Exception as e:
            print(f"  ⚠️ {api_name} недоступен: {e}")
            continue

    print("❌ Все API недоступны. Проверьте подключение к интернету.")
    sys.exit(1)

def main():
    print("🌐 Определение местоположения по IP...")
    my_lat, my_lon, my_country = get_ip_location()
    print(f"✅ Вы находитесь: {my_country} ({my_lat:.4f}, {my_lon:.4f})\n")

    distances = []
    for country, (cap_lat, cap_lon) in CAPITALS.items():
        if country == my_country:
            continue
        dist = haversine(my_lat, my_lon, cap_lat, cap_lon)
        distances.append((country, dist))

    distances.sort(key=lambda x: x[1])

    print(f'{"#":<3} {"Страна":<25} {"Расстояние (км)":<10}')
    print("-" * 45)
    for i, (country, dist) in enumerate(distances[:10], 1):
        print(f"{i:<3} {country:<25} {dist:.0f}")
    print("-" * 45)
    print("✅ Готово! Используйте список выше для настройки.")

if __name__ == "__main__":

#    по нему система находит корень после «Восстановить копию» в btrfs-assistant
#    (подтом переименовывается, а имя @ остаётся за восстановленным состоянием).

# 📋 Закрываем ESP от остальных пользователей. FAT не хранит права Unix — их
#    задают маски монтирования, а systemd-boot держит на ESP loader/random-seed.

#    выполнять через sudo. Это ожидаемое поведение, а не поломка.

#    прежнего GRUB_CMDLINE_LINUX_DEFAULT) добавляется сюда позже, в БЛОКЕ 20.
# ❗ rootflags=subvol=@ ОБЯЗАТЕЛЕН: корень монтируется явно по ИМЕНИ подтома.
#    Эта схема совместима с «Восстановить копию» в btrfs-assistant (подтома
#    переименовываются, а имя @ остаётся за восстановленным состоянием) и с
#    записями-снимками в меню загрузки. Схема с подтомом по умолчанию

# ℹ️ Используются systemd-хуки (нужны для plymouth и для будущих снимков):

#    sd-vconsole — заменяет keyboard/keymap/consolefont;
#    microcode — ОБЯЗАТЕЛЕН: с UKI микрокод упаковывается внутрь initramfs;
#    plymouth — ДО block, чтобы заставка появилась раньше всех запросов.
# ❗ Шифрования и LVM в этом макете нет — encrypt/sd-encrypt/lvm2 не добавляем.

sed -i 's|^#\?default_options=.*|default_options="--cmdline /etc/kernel/cmdline"|' /etc/mkinitcpio.d/linux-lts.preset

#    отдельный initramfs не собирается, всё упаковывается в UKI.
#    (В БЛОКЕ 13A мы вернём default_image обратно — он нужен записям-снимкам.)
# #------------------------------------------------------------------------------
# ШАГ 5: PLYMOUTH — ТЕМА ЗАСТАВКИ
# #------------------------------------------------------------------------------
# 🎨 bgrt — родная тема Arch (вариация spinner, удерживает логотип прошивки через
#    таблицу BGRT). В VirtualBox BGRT нет, поэтому будет только анимация — это норма.

# ШАГ 6: УСТАНОВКА systemd-boot И МЕНЮ ЗАГРУЗЧИКА
# #------------------------------------------------------------------------------

#    единственный пункт). editor no обязателен при Secure Boot.
cat > /boot/loader/loader.conf <<'EOF'

console-mode keep

EOF

# 🌿  [CHROOT] БЛОК 13A: СНИМКИ BTRFS В МЕНЮ systemd-boot

# ℹ️ Зачем: Полная замена связки grub-btrfs + grub-btrfsd + inotify-tools.
#    Снимки snapper появляются в меню загрузки как отдельные пункты.
# ⚠️ ПОЧЕМУ НЕ UKI: при включённом Secure Boot systemd-stub берёт параметры
#    ТОЛЬКО из секции .cmdline внутри UKI и игнорирует options из записи меню.
#    Подставить rootflags=subvol=@/.snapshots/N/snapshot в подписанный образ
#    нельзя, поэтому снимки грузятся обычной записью Type #1 (ядро + initramfs
#    отдельно). Ядро — PE-образ, прошивка проверяет его подпись, Secure Boot работает.
# 📌 Требование: ESP смонтирован в /boot (у нас так), чтобы systemd-boot мог
#    прочитать ядро и initramfs.
# ❗ Порядок: выполнять ПОСЛЕ БЛОКА 12 (нужны systemd-boot и пресет UKI).
#################################################################
# 🧹 Очистка экрана терминала
clear
# #------------------------------------------------------------------------------
# ШАГ 1: ВЕРНУТЬ СБОРКУ ОБЫЧНОГО INITRAMFS (нужен для записей-снимков)
# #------------------------------------------------------------------------------
# ℹ️ В БЛОКЕ 12 мы закомментировали default_image — всё упаковывалось в UKI.
#    Записям-снимкам нужен отдельный initramfs рядом с ядром на ESP.
sed -i 's|^#default_image=|default_image=|' /etc/mkinitcpio.d/linux-lts.preset
# #------------------------------------------------------------------------------
# ШАГ 2: ГЕНЕРАТОР ПУНКТОВ МЕНЮ
# #------------------------------------------------------------------------------
# ℹ️ Единственное исключение в макете: цикл по снимкам должен где-то храниться,
#    чтобы его могли вызывать хук pacman, path-юнит и таймер. Это НЕ скрипт
#    установки — установочные шаги по-прежнему выполняются командами.
# 💡 Что делает: берёт РЕАЛЬНЫЙ список подтомов (btrfs subvolume list /), отбирает
#    те, что похожи на снимки, и создаёт по одной записи меню на снимок.
#    Путь к снимкам не угадывается — поэтому работают и snapper, и btrfs-assistant,
#    и timeshift-подобные схемы. Шаблон отбора можно задать переменной PATTERN.
cat > /usr/local/bin/snapshot-entries <<'EOF'
#!/usr/bin/env bash
# Пункты меню systemd-boot для снимков btrfs (замена grub-btrfs).
# Список снимков берётся из РЕАЛЬНОГО списка подтомов, поэтому не важно,
# чем они созданы: snapper, btrfs-assistant или timeshift.
set -euo pipefail

command -v btrfs >/dev/null 2>&1 || exit 0           # не btrfs — тихо выходим

ESP="${ESP:-/boot}"                                  # для ESP в /boot/efi: ESP=/boot/efi
KERNEL="${KERNEL:-/vmlinuz-linux-lts}"               # пути ОТНОСИТЕЛЬНО ESP
INITRD="${INITRD:-/initramfs-linux-lts.img}"
ENTRIES="$ESP/loader/entries"

# Базовые параметры ядра берём из /etc/kernel/cmdline — тогда записи-снимков
# всегда совпадают с основной записью (quiet/splash включаются и выключаются
# вместе с ней). Свой rootflags= из базовой строки вычищаем — он будет своим.
ROOT_UUID="$(findmnt -no UUID /)"
BASE="$(sed 's/rootflags=[^ ]*//g; s/  */ /g; s/^ *//; s/ *$//' /etc/kernel/cmdline 2>/dev/null || true)"
[ -n "$BASE" ] || BASE="root=UUID=$ROOT_UUID rw"

# Подтом, из которого смонтирован ТЕКУЩИЙ КОРЕНЬ (например "@").
ROOT_SUB="$(findmnt -no SOURCE / | sed -n 's|.*\[\(.*\)\]|\1|p' | sed 's|^/||')"
[ -n "$ROOT_SUB" ] || ROOT_SUB="@"

# 🛑 КОГДА МЕНЮ ТРОГАТЬ НЕЛЬЗЯ (защита от удаления пунктов):
# 1) Загрузка ВНУТРИ снимка (откат через меню загрузки). В снимке каталог
#    .snapshots — пустая директория: без этого выхода генератор удалил бы
#    все пункты меню прямо во время такого сеанса (через 2 мин — таймер).
case "$ROOT_SUB" in
    */.snapshots/*/snapshot)       exit 0 ;;
    */timeshift-btrfs/snapshots/*) exit 0 ;;
esac
# 2) В текущем корне ВООБЩЕ нет каталога снимков. Так бывает в момент
#    «Восстановить копию» в btrfs-assistant: корень уже переименован в
#    @_backup_<время>, а дерево .snapshots переехало в новый корень.
#    Существующие пункты меню остаются верными — не удаляем их.
[ -d /.snapshots ] || [ -d /timeshift-btrfs ] || exit 0

# ❗ Отбираем только снимки КОРНЕВОГО подтома. Снимок другого подтома
#    (например @home/.snapshots/1/snapshot) грузить как корень НЕЛЬЗЯ.
#      snapper   → @/.snapshots/1/snapshot
#      timeshift → @/timeshift-btrfs/snapshots/2026-10-06_21-00-00/@
PATTERN="${PATTERN:-^${ROOT_SUB}/(\.snapshots/[0-9]+/snapshot|timeshift-btrfs/snapshots/[^/]+/[^/]+)$}"

mkdir -p "$ENTRIES"
find "$ENTRIES" -maxdepth 1 -name 'snapshot-*.conf' -delete

# ℹ️ ВАЖНО: "btrfs subvolume list /" печатает пути ПО-РАЗНОМУ в зависимости от
#    подтома по умолчанию — либо от корня ФС ("@/.snapshots/1/snapshot"),
#    либо от самого подтома (".snapshots/1/snapshot"). Поэтому каждый путь
#    приводим к виду от корня ФС: именно такой принимает параметр subvol=.
btrfs subvolume list / | sed 's/^.* path //' | sort -V | while read -r sub; do
    [ -n "$sub" ] || continue
    case "$sub" in
        "$ROOT_SUB"/*) ;;                        # уже от корня ФС
        *)             sub="$ROOT_SUB/$sub" ;;   # был от корневого подтома
    esac
    printf '%s\n' "$sub" | grep -qE "$PATTERN" || continue

    # номер снимка (snapper) либо контрольная сумма пути (другие схемы)
    num="$(printf '%s' "$sub" | grep -oE '/[0-9]+/' | tr -d '/' | head -1 || true)"
    [ -n "$num" ] || num="$(printf '%s' "$sub" | cksum | cut -d' ' -f1)"
    # путь в смонтированной системе — чтобы взять дату создания снимка
    mnt="/${sub#"$ROOT_SUB"/}"
    when="$(date -r "$mnt" '+%Y-%m-%d %H:%M' 2>/dev/null || echo 'дата неизвестна')"
    cat > "$ENTRIES/snapshot-$num.conf" <<ENTRY
title   Arch Linux — снимок $sub ($when)
linux   $KERNEL
initrd  $INITRD
options $BASE rootflags=subvol=$sub
ENTRY
done
EOF
chmod +x /usr/local/bin/snapshot-entries
# #------------------------------------------------------------------------------
# ШАГ 3: ПЕРВЫЙ ПРОГОН И ПРОВЕРКА
# #------------------------------------------------------------------------------
# 📋 Пересобираем initramfs (теперь собирается и UKI, и обычный образ)
mkinitcpio -P
# 📋 Генерируем пункты меню (на этом этапе снимков обычно ещё нет — конфиг
#    snapper создаётся позже в btrfs-assistant; пункты добавятся автоматически)
/usr/local/bin/snapshot-entries
# #------------------------------------------------------------------------------
# ШАГ 4: АВТООБНОВЛЕНИЕ ПОСЛЕ ТРАНЗАКЦИЙ PACMAN (замена grub-btrfsd)
# #------------------------------------------------------------------------------
# ℹ️ snap-pac создаёт снимки до/после транзакции — этот хук сразу обновляет меню.
cat > /usr/share/libalpm/hooks/zz-snapshot-entries.hook <<'EOF'
[Trigger]
Operation = Install
Operation = Upgrade
Operation = Remove
Type = Package
Target = *

[Action]
Description = Обновление пунктов меню systemd-boot для снимков btrfs
When = PostTransaction
Exec = /usr/local/bin/snapshot-entries
EOF
# #------------------------------------------------------------------------------
# ШАГ 5: МГНОВЕННАЯ РЕАКЦИЯ НА НОВЫЕ СНИМКИ (замена inotify из grub-btrfsd)
# #------------------------------------------------------------------------------
# ℹ️ path-юнит systemd сам следит за каталогом /.snapshots и сразу запускает
#    генератор: пункт меню появляется БЕЗ ожидания и без перезагрузки.
#    Пакет inotify-tools НЕ нужен — слежение встроено в systemd.
# 📌 ConditionPathIsDirectory нужен, чтобы юнит не считался упавшим, пока конфиг
#    snapper не создан и каталога /.snapshots ещё нет. После настройки snapper
#    в btrfs-assistant выполните один раз: sudo systemctl restart snapshot-entries.path
# 📌 sleep 2 в сервисе — защита от гонки: snapper сначала создаёт каталог
#    /.snapshots/N, и только потом внутри него подтом snapshot.
cat > /etc/systemd/system/snapshot-entries.service <<'EOF'
[Unit]
Description=Обновить пункты меню systemd-boot для снимков btrfs

[Service]
Type=oneshot
ExecStartPre=/usr/bin/sleep 2
ExecStart=/usr/local/bin/snapshot-entries
EOF
cat > /etc/systemd/system/snapshot-entries.path <<'EOF'
[Unit]
Description=Слежение за снимками btrfs и обновление меню systemd-boot
ConditionPathIsDirectory=/.snapshots

[Path]
PathChanged=/.snapshots
Unit=snapshot-entries.service

[Install]
WantedBy=multi-user.target
EOF
systemctl daemon-reload
systemctl enable --now snapshot-entries.path
# #------------------------------------------------------------------------------
# ШАГ 6: ТАЙМЕР — СТРАХОВКА (на случай, если path-юнит не сработал)
# #------------------------------------------------------------------------------
cat > /etc/systemd/system/snapshot-entries.timer <<'EOF'
[Unit]
Description=Периодическое обновление пунктов меню снимков btrfs

[Timer]
OnBootSec=2min
OnUnitActiveSec=1h

[Install]
WantedBy=timers.target
EOF
systemctl enable snapshot-entries.timer
# #------------------------------------------------------------------------------
# ШАГ 7: ПРОВЕРКА
# #------------------------------------------------------------------------------
clear
echo " 🔍 Пресет: default_image и default_uki должны быть БЕЗ # в начале:"
grep -E '^(default_image|default_uki)' /etc/mkinitcpio.d/linux-lts.preset
echo " "
echo " 🔍 На ESP есть UKI, ядро и initramfs (нужны записям-снимкам):"
ls -lh /boot/EFI/Linux/arch-linux-lts.efi /boot/vmlinuz-linux-lts /boot/initramfs-linux-lts.img
echo " "
echo " 🔍 Записи меню (основная + снимки; снимков пока может не быть — это норма):"
ls -1 /boot/loader/entries/
echo " "
echo " 🔍 Три механизма обновления меню (хук pacman, path-юнит, таймер):"
ls -1 /usr/share/libalpm/hooks/zz-snapshot-entries.hook
systemctl is-enabled snapshot-entries.path
systemctl is-enabled snapshot-entries.timer
echo " "

echo "## ✅ СНИМКИ BTRFS В МЕНЮ ЗАГРУЗКИ НАСТРОЕНЫ        ##"
echo "#####################################################"
echo " "

# 📌 КОГДА МЕНЮ ОБНОВЛЯЕТСЯ:
# • МГНОВЕННО — path-юнит snapshot-entries.path следит за /.snapshots и запускает
#   генератор сразу после создания снимка (в btrfs-assistant, snapper, snap-pac).
#   Перезагрузка не нужна: пункт появится в меню при следующем включении.
# • после любой транзакции pacman — хук zz-snapshot-entries.hook;
# • раз в час и через 2 мин после загрузки — таймер snapshot-entries.timer (страховка);
# • вручную: sudo snapshot-entries
# ⚠️ ЕСЛИ КОНФИГ SNAPPER СОЗДАН ПОСЛЕ УСТАНОВКИ: каталога /.snapshots на момент
#    установки ещё не было, поэтому один раз выполните
#    sudo systemctl restart snapshot-entries.path — дальше всё работает мгновенно.
# 🔍 Если пунктов всё равно нет — диагностика по шагам:
#    ls -d /.snapshots/*/         # существуют ли каталоги снимков вообще
#    sudo snapshot-entries        # скрипт с set -e покажет ошибку сразу
#    ls -1 /boot/loader/entries/  # создались ли файлы snapshot-N.conf
#    bootctl list                 # что реально видит systemd-boot
#    systemctl list-timers snapshot-entries.timer   # работает ли таймер
#
# 📌 Как этим пользоваться после установки:
# • Перезагрузка → в меню появятся пункты «Arch Linux — снимок N (дата)».
#   Если не успеваете выбрать за 3 секунды: sudo bootctl set-timeout 10
# • Откат основной системы: в btrfs-assistant выбрать снимок → «Восстановить
#   копию» → перезагрузка. Пункты меню при этом НЕ удаляются (защита в
#   генераторе). После перезагрузки проверьте: ls /.snapshots/ — если каталог
#   пуст, выполните ОДИН РАЗ БЛОК 13B: он вернёт дерево снимков в новый корень
#   (и пункты меню вместе с ним). Аварийный вариант — загрузиться в
#   пункт-снимок из меню и посмотреть состояние; меню при такой загрузке
#   тоже сохраняется. Команда snapper rollback в этой схеме НЕ
#   используется: она меняет подтом по умолчанию и конфликтует с
#   переименованием подтомов, которое выполняет btrfs-assistant.
# • Изменения внутри загруженного снимка НЕ попадают в основную систему.
#
# 🔍 Как убедиться, что загрузились ИМЕННО В СНИМОК (выполнять уже внутри него):
#    cat /proc/cmdline                        → rootflags=subvol=@/.snapshots/N/snapshot
#    findmnt -o SOURCE,TARGET,OPTIONS /       → /dev/sda2[/@/.snapshots/N/snapshot]
#    В основной системе там будет просто subvol=@ — это нормальный вывод,
#    а не признак поломки.
#
# ⚠️ ВАЖНО: /boot (ESP) у снимка и основной системы ОБЩИЙ. Находясь внутри снимка,
#    не обновляйте ядро и не запускайте mkinitcpio -P — файлы arch-linux-lts.efi
#    и initramfs-linux-lts.img перезапишутся сразу для всех пунктов меню.
#    Снимок нужен для осмотра и отката, а не для повседневной работы.
# 📌 Secure Boot: подписывать нужно только ядро (БЛОК 17 его подписывает).
#    initramfs — не PE-образ, подпись не требуется. Сами .conf-файлы не
#    подписываются: при editor no править их из меню нельзя, но физический
#    доступ к ESP позволяет — это общее свойство записей Type #1.
#################################################################




#################################################################
# 🌿  [СИСТЕМА] БЛОК 13B: ВОЗВРАТ ДЕРЕВА СНИМКОВ ПОСЛЕ ОТКАТА
#################################################################
# ℹ️ Когда нужен: после «Восстановить копию» в btrfs-assistant и перезагрузки
#    ls /.snapshots/ пуст — из меню загрузки пропали пункты снимков, а
#    snapper/btrfs-assistant не видят старую историю. Если в /.snapshots есть
#    нумерованные каталоги — всё в порядке, выполнять блок НЕ нужно.
# 📌 Почему так бывает: откат переименовывает корень (@ → @_backup_<время>)
#    и создаёт новый @ из снимка. Дерево .snapshots (вся история) живёт внутри @
#    и должно переехать в новый корень; иногда перенос не выполняется
#    (окно «Restore was successful but the migration of the nested subvolumes
#    failed»). Тогда история остаётся в @_backup_<время>, а в новом корне —
#    лишь пустой каталог .snapshots. Снимки НЕ удалены — блок переносит
#    дерево обратно, и пункты меню возвращаются.
# 📌 Выполнять из загруженной восстановленной системы от своего пользователя,
#    ОДИН раз после каждого отката через btrfs-assistant (блок сам проверит,
#    нужен ли ремонт).
# ❗ Если корневой раздел у вас не /dev/sda2 — замените в командах ниже.
#################################################################
# 🧹 Очистка экрана терминала
clear
# 🔍 Проверяем, где дерево снимков, и при необходимости переносим его в корень.
if ls -d /.snapshots/*/ >/dev/null 2>&1; then
  echo " ✅ Дерево снимков уже в корне — чинить нечего."
else
  # 📦 Монтируем верхний уровень ФС: в /mnt видны ВСЕ подтома.
  sudo mount -o subvolid=5 /dev/sda2 /mnt
  if ls -d /mnt/@/.snapshots/*/ >/dev/null 2>&1; then
    echo " ✅ Дерево уже в новом корне @ — станет видно после перезагрузки."
  else
    # 🔍 Ищем backup, в котором осталось дерево с нумерованными снимками.
    BK=""
    for d in /mnt/@_backup_*/; do ls -d "$d".snapshots/[0-9]*/ >/dev/null 2>&1 && BK="$d"; done
    if [ -n "$BK" ]; then
      echo " 🔍 Найден backup с деревом снимков: $BK"
      # 📦 Освобождаем место в новом корне: пустой каталог удаляется; если там
      #    уже успели создать новые снимки — он будет ПЕРЕИМЕНОВАН, а не удалён.
      sudo rmdir /mnt/@/.snapshots 2>/dev/null \
        || sudo mv /mnt/@/.snapshots "/mnt/@/.snapshots-new-$(date +%Y-%m-%d_%H%M%S)" 2>/dev/null
      # 📦 Переносим дерево снимков из backup в новый корень.
      sudo mv "${BK}.snapshots" /mnt/@/.snapshots
      echo " ✅ Дерево снимков возвращено в корень."
    else
      echo " ❗ Backup с деревом снимков не найден. Выполните и покажите вывод:"
      echo "    ls -d /mnt/@* ; sudo btrfs subvolume list /mnt | head -30"
    fi
  fi
  sudo umount /mnt
  # 📋 Заново вооружаем слежение за снимками и сразу пересобираем меню.
  sudo systemctl restart snapshot-entries.path 2>/dev/null
  sudo snapshot-entries
  # 🔍 Проверка: нумерованные снимки и пункты меню на месте.
  ls -d /.snapshots/*/ 2>/dev/null
  sudo ls -1 /boot/loader/entries/
fi
echo "#####################################################"
echo "## ✅ ДЕРЕВО СНИМКОВ ПРОВЕРЕНО / ВОЗВРАЩЕНО         ##"
echo "#####################################################"
#################################################################
# 📌 Старый корень остаётся в @_backup_<время>. Убедившись, что система
#    работает, его можно удалить (вкладка Subvolumes в btrfs-assistant, либо):
#      sudo mount -o subvolid=5 /dev/sda2 /mnt
#      sudo btrfs subvolume delete /mnt/@_backup_<время>
#      sudo umount /mnt
#    ❗ Смешанный (текущий) корень удалить нельзя — «Cannot delete mounted
#    subvolume» — поэтому только после перезагрузки в восстановленную систему.
#################################################################

#################################################################
# ℹ️ Зачем: Создание своих ключей и подпись всех загрузочных файлов.

#    с включённым Secure Boot НЕ ЗАГРУЗИТСЯ.
# 💡 VirtualBox: отдельную галочку дёргать не нужно. Пока в UEFI-хранилище нет
#    PK, прошивка в Setup Mode и Secure Boot фактически выключен (поэтому
#    live-ISO грузится, а запись ключей проходит). После записи PK прошивка
#    переходит в User Mode, и Secure Boot включается САМ при следующей загрузке.
# 📌 На реальном ПК: если sbctl enroll-keys ответит "Access denied" — зайдите в
#    BIOS/UEFI и очистите ключи Secure Boot (режим Setup), затем повторите.

# 🧹 Очистка экрана терминала
clear

# ШАГ 1: ФИНАЛЬНАЯ ПЕРЕСБОРКА UKI И INITRAMFS (строго ДО подписи!)
# #------------------------------------------------------------------------------
mkinitcpio -P
# #------------------------------------------------------------------------------
# ШАГ 2: КЛЮЧИ SECURE BOOT
# #------------------------------------------------------------------------------
# 📋 Создать свои ключи (хранятся в /var/lib/sbctl)
sbctl create-keys
# 📋 Записать ключи в прошивку. Флаг -m сохраняет ключи Microsoft — они нужны
#    для видеокарты с подписанным ROM, dual-boot и обновления прошивки.
sbctl enroll-keys -m
# #------------------------------------------------------------------------------
# ШАГ 3: ПОДПИСАТЬ ВСЁ, ЧТО УЧАСТВУЕТ В ЗАГРУЗКЕ
# #------------------------------------------------------------------------------
# 📋 systemd-boot лежит в двух местах: основном и запасном (EFI/BOOT).
#    VirtualBox грузится именно по запасному пути — подписывать ОБА обязательно.
sbctl sign -s /boot/EFI/systemd/systemd-bootx64.efi
sbctl sign -s /boot/EFI/BOOT/BOOTX64.EFI
# 📋 UKI — основной пункт меню (ядро + initramfs + параметры в одном файле)
sbctl sign -s /boot/EFI/Linux/arch-linux-lts.efi
# 📋 Ядро — нужно записям-снимкам из БЛОКА 13A и как запасной путь загрузки
sbctl sign -s /boot/vmlinuz-linux-lts
# 📋 Прочие .efi на ESP подписываем ЯВНО: sbctl sign-all только переподписывает
#    уже известные ему файлы из базы, НОВЫЕ он не находит. Пакет memtest86+-efi
#    кладёт свой образ на ESP, поэтому без явной подписи sbctl verify будет
#    ругаться, а запуск memtest из меню при включённом SB даст "Access denied".
[ -f /boot/memtest86+/memtest.efi ] && sbctl sign -s /boot/memtest86+/memtest.efi
# 📋 Переподписать всё, что уже есть в базе sbctl (страховка)
sbctl sign-all
# #------------------------------------------------------------------------------
# ШАГ 4: ОБЯЗАТЕЛЬНАЯ ПРОВЕРКА
# #------------------------------------------------------------------------------
# ✅ В выводе НЕ ДОЛЖНО быть строк "✗" или "is not signed"
sbctl verify

echo " 🔍 Список подписанных файлов:"
sbctl list-files
echo " "
echo "#####################################################"
echo "## ✅ SECURE BOOT НАСТРОЕН                          ##"
echo "#####################################################"
echo " "

exit

# ℹ️ Зачем: попасть в установленную систему с live-ISO и починить загрузку.

#    Настройки → Система → Материнская плата → снять «Enable Secure Boot».
#    Live-ISO не подписан вашими ключами, и при включённом SB прошивка его
#    отбракует. Ключи при этом НЕ ТЕРЯЮТСЯ (они в NVRAM виртуальной машины) —
#    после ремонта просто верните галочку, заново ничего записывать не нужно.
# 📌 Также проверьте: Настройки → Носители (ISO подключён) и Порядок загрузки
#    («Оптический» включён и стоит выше диска). В момент логотипа VirtualBox
#    можно нажать F12 и выбрать устройство вручную.
# 📌 Если ВМ попала в UEFI Interactive Shell:
#    map -r   →   FS1:   →   \EFI\BOOT\BOOTX64.EFI
#################################################################
# 🧹 Очистка экрана терминала
clear
# #------------------------------------------------------------------------------
# ШАГ 1: НАЙТИ ДИСК И ПРОВЕРИТЬ ПОДТОМ ПО УМОЛЧАНИЮ
# #------------------------------------------------------------------------------
lsblk -f
# 📋 Монтируется подтом ПО УМОЛЧАНИЮ — по его содержимому виден диагноз
mount /dev/sda2 /mnt
ls /mnt
btrfs subvolume get-default /mnt
# ℹ️ В этом макете корень монтируется ЯВНО по имени (subvol=@), а подтом по
#    умолчанию остаётся toplevel. Поэтому ls /mnt ДОЛЖЕН показать
#    "@ @home @log @pkg" — это норма, а не признак поломки.
# ℹ️ Если ls показал содержимое обычной системы (bin, etc, usr, var) — значит,
#    смонтировался не тот подтом: проверьте fstab и cmdline в шаге 4.
# #------------------------------------------------------------------------------
# ШАГ 2: ПРОВЕРИТЬ, ЧТО КОРЕНЬ УКАЗАН ПО ИМЕНИ ПОДТОМА
# #------------------------------------------------------------------------------
# ❗ В схеме «по имени» корень ищется по subvol=@ в fstab и rootflags=subvol=@
#    в параметрах ядра. Подтом по умолчанию НЕ трогаем: btrfs-assistant
#    выполняет откат переименованием подтомов, и смена default сломает его.
grep -E '[[:space:]]/[[:space:]]' /etc/fstab
grep rootflags /etc/kernel/cmdline
umount /mnt
# #------------------------------------------------------------------------------
# ШАГ 3: СМОНТИРОВАТЬ СИСТЕМУ И ВОЙТИ В CHROOT
# #------------------------------------------------------------------------------
mount -o subvol=@ /dev/sda2 /mnt
mount -o subvol=@home /dev/sda2 /mnt/home
mount -o subvol=@log /dev/sda2 /mnt/var/log
mount -o subvol=@pkg /dev/sda2 /mnt/var/cache/pacman/pkg
mount /dev/sda1 /mnt/boot
arch-chroot /mnt
# #------------------------------------------------------------------------------
# ШАГ 4: ОБЯЗАТЕЛЬНЫЕ ДЕЙСТВИЯ ВНУТРИ CHROOT
# #------------------------------------------------------------------------------
# 🔑 1) Пароль root. Без него ЛЮБОЙ сбой = тупик: аварийная консоль пишет
#       "Cannot open access to console, the root account is locked" и не пускает.
passwd
# 📋 2) Посмотреть, что реально записано
cat /etc/fstab
cat /etc/kernel/cmdline
# 📋 3) ВРЕМЕННО отключить тихую загрузку — иначе причину сбоя не увидеть
sed -i 's/ quiet splash loglevel=3//' /etc/kernel/cmdline
# 📋 4) Проверить fstab: у корня ДОЛЖЕН быть subvol=/@ (и у @home, @log, @pkg
#       свои subvol=). Если у корня его нет — вернуть:
grep -qE '([[:space:]]/[[:space:]].*subvol=/@)' /etc/fstab || \
  sed -i '\|[[:space:]]/[[:space:]]|s|\(btrfs[[:space:]]\+rw[^[:space:]]*\)|\1,subvol=/@|' /etc/fstab
# 📋 4б) Проверить cmdline: должна быть строка rootflags=subvol=@
grep -q 'rootflags=subvol=@' /etc/kernel/cmdline || \
  sed -i 's|\(root=UUID=[^ ]* rw\)|\1 rootflags=subvol=@|' /etc/kernel/cmdline
# 📋 Закрываем ESP от остальных пользователей. FAT не хранит права Unix — их
#    задают маски монтирования, а systemd-boot держит на ESP loader/random-seed.
#    Без этого bootctl пишет: "Mount point '/boot' which backs the random seed
#    file is world accessible, which is a security hole!". Прошивке маски не
#    мешают: она читает FAT напрямую, минуя права доступа.
# ⚠️ Побочный эффект: после этого обычный пользователь НЕ СМОЖЕТ читать /boot —
#    команды вроде ls /boot, cat /boot/loader/loader.conf, bootctl, sbctl нужно
#    выполнять через sudo. Это ожидаемое поведение, а не поломка.
sed -i '\|[[:space:]]/boot[[:space:]]|{s|fmask=[0-9]*|fmask=0177|; s|dmask=[0-9]*|dmask=0077|}' /etc/fstab
# #------------------------------------------------------------------------------
# ШАГ 5: ПЕРЕСОБРАТЬ, ПОДПИСАТЬ И ПЕРЕЗАГРУЗИТЬ
# #------------------------------------------------------------------------------
mkinitcpio -P
sbctl sign-all 2>/dev/null || true
sbctl verify 2>/dev/null || true
exit
umount -R /mnt
reboot

# 📋 Вернуть Secure Boot (VirtualBox): Настройки → Система → Материнская плата

# #------------------------------------------------------------------------------

#    нужные таймеры. Какой конфиг и какое расписание — выбирает пользователь.
# ❗ Конфиг нужен именно для "/", а НЕ для "/home". Снимок другого подтома

#    они не появятся. Конфиг для /home при этом можно оставить: он полезен для
#    восстановления файлов, просто не участвует в загрузке.

sudo snapper list-configs

#    каталог /.snapshots появился уже после загрузки, и path-юнит его не видел.
sudo systemctl restart snapshot-entries.path
systemctl is-active snapshot-entries.path
# 📋 Проверка мгновенного попадания снимка в меню (конфиг указываем явно):

sudo ls -1 /boot/loader/entries/

# ✅ В строке options должно быть rootflags=subvol=@/.snapshots/N/snapshot —

# 📋 quiet — гасит лог, splash — включает заставку (нужны ОБА параметра),
#    loglevel=3 — не выводить на консоль сообщения уровня err и ниже.
sudo sed -i 's/$/ quiet splash loglevel=3/' /etc/kernel/cmdline
cat /etc/kernel/cmdline
# 📋 Пересборка UKI. Подпись ставится автоматически (post-хук sbctl внутри
#    mkinitcpio), sign-all — для страховки.
sudo mkinitcpio -P
sudo sbctl sign-all
sudo sbctl verify
# 📋 Перезагрузка — вместо логов появится заставка
sudo reboot
